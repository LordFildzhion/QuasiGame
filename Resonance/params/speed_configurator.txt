; Конфигурация SPEED + EMG Interpreter + QuasiGame (UTF-8).
; Меняйте значения справа от =. Изменения применятся после перезапуска SPEED / Prepare.
; Номера каналов начинаются с 1. Значение 0 отключает соответствующую руку.

[source]
; Имя потока для справки; фактический вход выбирается в Resonance Configurator.
; LSL уже подключён в Resonance, никакой внешний LSL-мост не нужен.
service = Native_LSL
stream = NVX136_Data
sampling_rate_hz = 500
; Единицы данных, которые отдаёт NATIVE Resonance LSL поток: V, mV, uV.
; Уточните фактические единицы вашего NVX.
input_units = V

[channels]
; Первый канал = 1, а не 0.
left_channel = 0
right_channel = 2
left_sensor_name = left_p
right_sensor_name = right_p

[filter]
; Полоса ЭМГ: high_hz должна быть меньше sampling_rate_hz / 2.
low_hz = 20
high_hz = 200
; Режектор 50 Гц: 0 отключит notch (адаптивное подавление 50 Гц останется).
notch_hz = 50
notch_q = 30
order = 4
; Короткая огибающая нужна для обнаружения микро-сокращений; прежние 200 мс их размазывали.
rms_window_ms = 20
attack_ms = 8
release_ms = 25

[normalization]
; baseline можно включать только для исходных каналов 64/66 и точно той же записи.
; manual: новые каналы/частоты. Старая калибровка НЕ применяется.
; Выход: относительная активность от 0 до 1, НЕ процент MVC и НЕ сила.
threshold_mode = manual
; В manual пороги RMS задаются в мкВ; 0 отключает вычитание порога.
; Перед серьёзным использованием настройте пороги по записям покоя и нагрузки.
left_threshold_uv = 0
right_threshold_uv = 0
rolling_window_seconds = 10

[bridge]
; Транспорт отделён от игрового движка. Bridge отправляет одно UDP-сообщение на сокращение.
enabled = true
host = 127.0.0.1
port = 8765

[event_detection]
; Первые 2 секунды после Start SPEED: обе руки расслаблены, оцениваем шум покоя.
calibration_seconds = 2.0
; Порог включения в долях робастного sigma шума. Начальное значение снижает ложные события от случайных колебаний.
; Меньше = лучше ловит микро-сокращения, но выше риск ложных срабатываний.
onset_sigma = 1.0
; Гистерезис отпускания: должно быть меньше onset_sigma.
release_sigma = 0.5
; Минимальная длительность сокращения. 20 мс при 500 Гц = 10 отсчётов; короткий одиночный шумовой пик не считается событием.
min_contraction_ms = 20
; Длительность ниже release threshold для завершения события.
release_ms = 28
; Защита от повторного срабатывания внутри одного сокращения/дрожания.
refractory_ms = 65
; Уровень 1.0 достигается приблизительно при пике в 14 sigma над покоем.
full_scale_sigma = 14
; Медленное сопровождение дрейфа фона только в состоянии покоя.
baseline_tracking_seconds = 12
minimum_noise_floor = 1e-12
